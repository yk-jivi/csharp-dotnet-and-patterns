Book.cs
using System.Collections.Generic;
using System.ComponentModel;
using System.ComponentModel.DataAnnotations;
using System.ComponentModel.DataAnnotations.Schema;
using System.Runtime.CompilerServices;

namespace BookStore
{
    public class Book : INotifyPropertyChanged
    {
        private int _id;
        private string _title = "";
        private string _author = "";
        private int _year;
        private int _pages;
        private decimal _price;
        private string _description = "";
        private string _imagePath = "";
        private int _rating;
        private int _selectedQuantity = 1;

        [Key]
        public int Id
        {
            get => _id;
            set { _id = value; OnPropertyChanged(); }
        }

        public string Title
        {
            get => _title;
            set { _title = value; OnPropertyChanged(); }
        }

        public string Author
        {
            get => _author;
            set { _author = value; OnPropertyChanged(); }
        }

        public decimal Price
        {
            get => _price;
            set { _price = value; OnPropertyChanged(); }
        }

        public int Year
        {
            get => _year;
            set { _year = value; OnPropertyChanged(); }
        }

        public int Pages
        {
            get => _pages;
            set { _pages = value; OnPropertyChanged(); }
        }

        public string Description
        {
            get => _description;
            set { _description = value; OnPropertyChanged(); }
        }

        [NotMapped]
        public string ImagePath
        {
            get => _imagePath;
            set { _imagePath = value; OnPropertyChanged(); }
        }

        [Column("ImageData")]
        public byte[]? ImageBytes { get; set; }

        // Связь с категорией
        public int? CategoryId { get; set; }

        public virtual Category? CategoryEntity { get; set; }

        // Для отображения в UI
        [NotMapped]
        public string Category
        {
            get => CategoryEntity?.Name ?? "Без категории";
            set { }
        }

        [NotMapped]
        public int Rating
        {
            get => _rating;
            set { _rating = value; OnPropertyChanged(); }
        }

        [NotMapped]
        public int SelectedQuantity
        {
            get => _selectedQuantity;
            set { _selectedQuantity = value; OnPropertyChanged(); }
        }

        public event PropertyChangedEventHandler? PropertyChanged;
        protected void OnPropertyChanged([CallerMemberName] string name = "")
        {
            PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(name));
        }
    }
}

using System.ComponentModel.DataAnnotations;
using System.ComponentModel.DataAnnotations.Schema;

namespace BookStore
{
    [Table("OrderItems")]
    public class OrderItem
    {
        [Key]
        public int Id { get; set; }

        public int OrderId { get; set; }
        [ForeignKey("OrderId")]
        public virtual Order Order { get; set; } = null!;

        public int BookId { get; set; }
        [ForeignKey("BookId")]
        public virtual Book Book { get; set; } = null!;

        public int Quantity { get; set; }
        public decimal UnitPrice { get; set; }
    }
}

CartItem.cs
using System.ComponentModel;
using System.Runtime.CompilerServices;

namespace BookStore
{
    public class CartItem : INotifyPropertyChanged
    {
        private Book? _book;
        private int _quantity = 1;

        public Book Book
        {
            get { return _book!; }
            set { _book = value; OnPropertyChanged(); }
        }

        public int Quantity
        {
            get { return _quantity; }
            set
            {
                _quantity = value;
                OnPropertyChanged();
                OnPropertyChanged(nameof(TotalPrice));
            }
        }

        public decimal TotalPrice
        {
            get { return (Book?.Price ?? 0) * Quantity; }
        }

        public event PropertyChangedEventHandler? PropertyChanged;
        protected void OnPropertyChanged([CallerMemberName] string name = null!)
        {
            PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(name));
        }
    }
}

        private bool _isFiltering = false;

        public async void FilterBooks()
        {
            if (_isFiltering) return;

            if (books == null || CategoryFilter == null || SortFilter == null || SearchBox == null)
                return;

            _isFiltering = true;

            try
            {
                // Сохраняем ID выбранной книги до фильтрации
                int? selectedBookId = SelectedBook?.Id;

                string searchText = SearchBox.Text.Trim().ToLower();
                var resource = TryFindResource("SearchPlaceholder");
                string searchPlaceholder = resource?.ToString()?.ToLower() ?? "";

                if (searchText == searchPlaceholder)
                    searchText = "";

                string selectedCategory = null;
                if (CategoryFilter.SelectedIndex > 0)
                {
                    var selectedItem = CategoryFilter.SelectedItem as ComboBoxItem;
                    selectedCategory = selectedItem?.Content?.ToString();
                }

                string sortBy = null;
                bool ascending = true;

                switch (SortFilter.SelectedIndex)
                {
                    case 1: // Цена по возрастанию
                        sortBy = "price";
                        ascending = true;
                        break;
                    case 2: // Цена по убыванию
                        sortBy = "price";
                        ascending = false;
                        break;
                    case 3: // По названию
                        sortBy = "title";
                        ascending = true;
                        break;
                }

                // Используем асинхронный метод с фильтрацией на стороне БД
                var filtered = await _db.GetFilteredBooksAsync(
                    searchText,
                    selectedCategory,
                    sortBy,
                    ascending);

                filteredBooks.Clear();
                foreach (var book in filtered)
                {
                    filteredBooks.Add(book);
                }

                // Восстанавливаем выбранную книгу
                if (selectedBookId.HasValue)
                {
                    var restoredSelection = filteredBooks.FirstOrDefault(b => b.Id == selectedBookId.Value);
                    SelectedBook = restoredSelection;

                    if (restoredSelection != null && BooksListView != null)
                    {
                        BooksListView.SelectedItem = restoredSelection;
                    }
                }

                if (BooksListView.ItemsSource != filteredBooks)
                {
                    BooksListView.ItemsSource = filteredBooks;
                }
                else
                {
                    BooksListView.Items.Refresh();
                }

                if (TileViewPanel.Visibility == Visibility.Visible)
                {
                    UpdateTileView();
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ошибка фильтрации: {ex.Message}", "Ошибка");
            }
            finally
            {
                _isFiltering = false;
            }
        }

        private void AddToCart(Book book)
        {
            var existingItem = cart.FirstOrDefault(item => item.Book.Id == book.Id);
            if (existingItem != null)

                existingItem.Quantity += book.SelectedQuantity;

            CartListView.Items.Refresh();

        public void ChangeTheme(string themeName)
        {
            try
            {
                var themeDict = new ResourceDictionary();
                themeDict.Source = new Uri($"{themeName}.xaml", UriKind.Relative);

                var dictionariesToRemove = new List<ResourceDictionary>();
                foreach (var dict in Application.Current.Resources.MergedDictionaries)
                {
                    if (dict.Source != null && (dict.Source.ToString().Contains("Theme") ||
                        dict.Source.ToString().Contains("DefaultTheme") ||
                        dict.Source.ToString().Contains("MochaTheme") ||
                        dict.Source.ToString().Contains("BlueTheme")))
                    {
                        dictionariesToRemove.Add(dict);
                    }
                }

                foreach (var dict in dictionariesToRemove)
                {
                    Application.Current.Resources.MergedDictionaries.Remove(dict);
                }

                Application.Current.Resources.MergedDictionaries.Add(themeDict);

                this.Resources = null;
                this.Resources = Application.Current.Resources;
                this.Background = (Brush)Application.Current.Resources["BackgroundBrush"];

                if (HeaderBorder != null)
                {
                    HeaderBorder.Background = (Brush)Application.Current.Resources["HeaderBrush"];
                }

                foreach (Window window in Application.Current.Windows)
                {
                    if (window != this && window is PersonalAccountWindow accountWindow)
                    {
                        accountWindow.UpdateTheme();
                    }
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ошибка при смене темы: {ex.Message}");
            }
        }

        private void EditBookBtn_Click(object sender, RoutedEventArgs e)
        {
            // Получаем книгу из CommandParameter если есть, иначе из SelectedBook
            Book bookToEdit = null;

            if (sender is Button button && button.CommandParameter is Book parameterBook)
            {
                bookToEdit = parameterBook;
            }
            else if (SelectedBook != null)
            {
                bookToEdit = SelectedBook;
            }

            if (bookToEdit == null)
            {
                MessageBox.Show("Выберите книгу для редактирования!", "Внимание",
                    MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            selectedBookForEdit = bookToEdit;
            DialogTitle.Text = (string)Application.Current.Resources["EditTitle"];

            DialogTitleInput.Text = selectedBookForEdit.Title;
            DialogAuthorInput.Text = selectedBookForEdit.Author;

            // Устанавливаем категорию
            bool categoryFound = false;
            for (int i = 0; i < DialogCategoryInput.Items.Count; i++)
            {
                var item = DialogCategoryInput.Items[i] as ComboBoxItem;
                if (item != null && item.Content.ToString() == selectedBookForEdit.Category)
                {
                    DialogCategoryInput.SelectedIndex = i;
                    categoryFound = true;
                    break;
                }
            }

            if (!categoryFound)
            {
                DialogCategoryInput.SelectedIndex = 0; // Выбираем "Выберите категорию"
            }

            DialogYearInput.Text = selectedBookForEdit.Year > 0 ? selectedBookForEdit.Year.ToString() : "";
            DialogPagesInput.Text = selectedBookForEdit.Pages > 0 ? selectedBookForEdit.Pages.ToString() : "";
            DialogPriceInput.Text = selectedBookForEdit.Price.ToString();

            // Сбрасываем временные байты изображения
            currentSelectedImageBytes = null;

            BookDialog.Visibility = Visibility.Visible;
        }

        private async void DeleteBookBtn_Click(object sender, RoutedEventArgs e)
        {
            Book bookToDelete = null;

            if (sender is Button button && button.CommandParameter is Book parameterBook)
            {

        private async void DialogSaveBtn_Click(object sender, RoutedEventArgs e)
        {
            try
            {
                // Валидация полей
                if (string.IsNullOrWhiteSpace(DialogTitleInput.Text))
                {
                    MessageBox.Show("Введите название книги!", "Ошибка");
                    return;
                }

                if (DialogCategoryInput.SelectedIndex <= 0)
                {
                    MessageBox.Show("Выберите категорию!", "Ошибка");
                    return;
                }

                if (!decimal.TryParse(DialogPriceInput.Text, out decimal price) || price <= 0)
                {
                    MessageBox.Show("Введите корректную цену!", "Ошибка");
                    return;
                }

                // Получаем ID категории
                string selectedCatName = (DialogCategoryInput.SelectedItem as ComboBoxItem)?.Content.ToString();
                int categoryId = selectedCatName switch
                {
                    "Романы" => 1,
                    "Поэзия" => 2,
                    "Драма" => 3,
                    "Фэнтези" => 4,
                    "Классика" => 5,
                    _ => 1
                };

                // Сбор данных
                int.TryParse(DialogYearInput.Text, out int year);
                int.TryParse(DialogPagesInput.Text, out int pages);

                var bookData = new Book
                {
                    Title = DialogTitleInput.Text.Trim(),
                    Author = DialogAuthorInput.Text.Trim(),
                    Price = price,
                    Year = year,
                    Pages = pages,
                    ImageBytes = this.currentSelectedImageBytes
                };

                if (selectedBookForEdit == null)
                {
                    // Добавление (АСИНХРОННО)
                    await _db.AddBookWithTransactionAsync(bookData, categoryId);
                    MessageBox.Show("Книга добавлена!");
                }
                else
                {
                    // Редактирование (АСИНХРОННО)
                    bookData.Id = selectedBookForEdit.Id;
                    await _db.UpdateBookWithTransactionAsync(bookData, categoryId);
                    MessageBox.Show("Книга обновлена!");
                }

                // Перезагружаем данные
                await LoadBooksFromDatabaseAsync();

                // Закрываем диалог
                this.currentSelectedImageBytes = null;
                BookDialog.Visibility = Visibility.Collapsed;
                selectedBookForEdit = null;
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ошибка: {ex.Message}", "Ошибка");
            }
        }

        private async Task LoadBooksFromDatabaseAsync()

            }

            {
                MessageBox.Show($"Ошибка загрузки: {ex.Message}");
            }
        }

using System.Windows;
using System.Windows.Controls;

namespace BookStore
{
    public partial class NumberBox : UserControl
    {
        public static readonly DependencyProperty ValueProperty =
            DependencyProperty.Register("Value", typeof(int), typeof(NumberBox),
                new FrameworkPropertyMetadata(1, OnValueChanged, CoerceValue), ValidateValue);

        public static readonly DependencyProperty MinProperty =
            DependencyProperty.Register("Min", typeof(int), typeof(NumberBox),
                new PropertyMetadata(1, null, CoerceMin));

        public static readonly DependencyProperty MaxProperty =
            DependencyProperty.Register("Max", typeof(int), typeof(NumberBox),
                new PropertyMetadata(10, null, CoerceMax));

        public static readonly RoutedEvent ValueChangedEvent =
            EventManager.RegisterRoutedEvent("ValueChanged", RoutingStrategy.Tunnel,
                typeof(RoutedEventHandler), typeof(NumberBox));

        public event RoutedEventHandler ValueChanged
        {
            add { AddHandler(ValueChangedEvent, value); }
            remove { RemoveHandler(ValueChangedEvent, value); }
        }

        public int Value
        {
            get { return (int)GetValue(ValueProperty); }
            set { SetValue(ValueProperty, value); }
        }

        public int Min
        {
            get { return (int)GetValue(MinProperty); }
            set { SetValue(MinProperty, value); }
        }

        public int Max
        {
            get { return (int)GetValue(MaxProperty); }
            set { SetValue(MaxProperty, value); }
        }

        public NumberBox()
        {
            InitializeComponent();
        }

        private static bool ValidateValue(object value)
        {
            int val = (int)value;
            return val >= 0;
        }

        private static object CoerceValue(DependencyObject d, object baseValue)
        {
            var control = (NumberBox)d;
            int val = (int)baseValue;

            if (val < control.Min) return control.Min;
            if (val > control.Max) return control.Max;
            return val;
        }

        private static object CoerceMin(DependencyObject d, object baseValue)
        {
            var control = (NumberBox)d;
            int min = (int)baseValue;
            if (min > control.Max) return control.Max;
            if (min < 0) return 0;
            return min;
        }

        private static object CoerceMax(DependencyObject d, object baseValue)
        {
            var control = (NumberBox)d;
            int max = (int)baseValue;
            if (max < control.Min) return control.Min;
            return max;
        }

        private static void OnValueChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
        {
            var control = (NumberBox)d;
            control.RaiseEvent(new RoutedEventArgs(ValueChangedEvent, control));
        }

        private void Increase_Click(object sender, RoutedEventArgs e)
        {
            if (Value < Max) Value++;

            RaiseEvent(new RoutedEventArgs(ValueChangedEvent, sender));
        }

        private void Decrease_Click(object sender, RoutedEventArgs e)
        {
            if (Value > Min) Value--;

            RaiseEvent(new RoutedEventArgs(ValueChangedEvent, sender));
        }
    }

using System;
using System.Collections.ObjectModel;
using System.Data;
using System.Linq;
using System.Threading.Tasks;
using BookStore.DAL;
using System.Data.Entity;

namespace BookStore
{
    // Сервис для работы с данными через паттерны Repository + Unit of Work
    public class EFService
    {

        // Получить все книги с категориями (синхронно, для обратной совместимости)
        public ObservableCollection<Book> GetAllBooks()
        {
            using (var uow = new UnitOfWork())
            {
                var books = uow.Books.Query()
                    .Include(b => b.CategoryEntity)
                    .ToList();
                return new ObservableCollection<Book>(books);
            }
        }

        // Получить все книги асинхронно
        public async Task<ObservableCollection<Book>> GetAllBooksAsync()
        {
            using (var uow = new UnitOfWork())
            {
                var books = await uow.Books.Query()
                    .Include(b => b.CategoryEntity)
                    .ToListAsync();
                return new ObservableCollection<Book>(books);
            }
        }

        // Получить отфильтрованные книги (для MainWindow.FilterBooks)
        public async Task<ObservableCollection<Book>> GetFilteredBooksAsync(
            string searchText = null,
            string category = null,
            string sortBy = null,
            bool ascending = true)
        {
            using (var uow = new UnitOfWork())
            {
                // Получаем IQueryable из репозитория
                IQueryable<Book> query = uow.Books.Query()
                    .Include(b => b.CategoryEntity);

                // Поиск по нескольким полям
                if (!string.IsNullOrWhiteSpace(searchText))
                {
                    searchText = searchText.ToLower();
                    query = query.Where(b =>
                        b.Title.ToLower().Contains(searchText) ||
                        b.Author.ToLower().Contains(searchText) ||
                        b.Description.ToLower().Contains(searchText));
                }

                // Фильтрация по категории
                if (!string.IsNullOrWhiteSpace(category) && category != "Все категории")
                {
                    query = query.Where(b => b.CategoryEntity.Name == category);
                }

                // Сортировка
                if (!string.IsNullOrWhiteSpace(sortBy))
                {
                    query = sortBy.ToLower() switch
                    {
                        "title" => ascending
                            ? query.OrderBy(b => b.Title)
                            : query.OrderByDescending(b => b.Title),
                        "author" => ascending
                            ? query.OrderBy(b => b.Author)
                            : query.OrderByDescending(b => b.Author),
                        "price" => ascending
                            ? query.OrderBy(b => b.Price)
                            : query.OrderByDescending(b => b.Price),
                        "year" => ascending
                            ? query.OrderBy(b => b.Year)
                            : query.OrderByDescending(b => b.Year),
                        _ => query
                    };
                }

                var result = await query.ToListAsync();
                return new ObservableCollection<Book>(result);
            }
        }

        // Добавить книгу (синхронно)
        public void AddBook(Book book, int categoryId)
        {
            using (var uow = new UnitOfWork())
            {
                try
                {
                    uow.BeginTransaction();

                    book.CategoryId = categoryId;
                    uow.Books.Add(book);
                    uow.Save();
                    uow.Commit();
                }
                catch (Exception)
                {
                    uow.Rollback();
                    throw;
                }
            }
        }

        // Добавить книгу асинхронно (с транзакцией)
        public async Task AddBookWithTransactionAsync(Book book, int categoryId)
        {
            using (var uow = new UnitOfWork())
            {
                try
                {
                    uow.BeginTransaction();

                    book.CategoryId = categoryId;
                    uow.Books.Add(book);
                    await uow.SaveAsync();
                    uow.Commit();
                }
                catch (Exception)
                {

                    throw;
                }
            }
        }

        // Обновить книгу (синхронно)
        public void UpdateBook(Book book, int categoryId)
        {
            using (var uow = new UnitOfWork())
            {
                try
                {
                    uow.BeginTransaction();

                    var existingBook = uow.Books.Query()
                        .FirstOrDefault(b => b.Id == book.Id);

                    if (existingBook != null)
                    {
                        existingBook.Title = book.Title;
                        existingBook.Author = book.Author;
                        existingBook.Price = book.Price;
                        existingBook.Year = book.Year;
                        existingBook.Pages = book.Pages;
                        existingBook.CategoryId = categoryId;
                        existingBook.Description = book.Description;

                        if (book.ImageBytes != null)
                        {
                            existingBook.ImageBytes = book.ImageBytes;
                        }

                        uow.Books.Update(existingBook);
                        uow.Save();
                        uow.Commit();
                    }
                }
                catch (Exception)
                {
                    uow.Rollback();
                    throw;
                }
            }
        }

        // Обновить книгу асинхронно (с транзакцией)
        public async Task UpdateBookWithTransactionAsync(Book book, int categoryId)
        {

            {

                {

                    uow.BeginTransaction();

                    uow.Books.Delete(id);
                    await uow.SaveAsync();
                    uow.Commit();
                }
                catch (Exception)
                {
                    uow.Rollback();
                    throw;
                }
            }
        }

        // Создать заказ (синхронно)
        public int CreateOrder(ObservableCollection<CartItem> cart)
        {
            using (var uow = new UnitOfWork())
            {
                try
                {
                    uow.BeginTransaction();

                    // Создаем заказ
                    var order = new Order
                    {
                        OrderDate = DateTime.Now,
                        TotalAmount = cart.Sum(item => item.TotalPrice)
                    };
                    uow.Orders.Add(order);
                    uow.Save();

                    // Добавляем элементы заказа
                    foreach (var item in cart)
                    {
                        var orderItem = new OrderItem
                        {
                            OrderId = order.Id,
                            BookId = item.Book.Id,
                            Quantity = item.Quantity,
                            UnitPrice = item.Book.Price
                        };
                        uow.OrderItems.Add(orderItem);
                    }

                    uow.Save();
                    uow.Commit();
                    return order.Id;
                }
                catch (Exception)
                {
                    uow.Rollback();
                    throw;
                }
            }
        }

        // Создать заказ асинхронно
        public async Task<int> CreateOrderAsync(ObservableCollection<CartItem> cart)
        {

                    }

                }
                catch (Exception)
                {

                    throw;
                }
            }
        }

        // Получить таблицу категорий
        public DataTable GetCategoriesTable()
        {
            using (var uow = new UnitOfWork())
            {
                var dt = new DataTable();
                dt.Columns.Add("Id", typeof(int));
                dt.Columns.Add("Name", typeof(string));

                var categories = uow.Categories.Query().ToList();
                foreach (var c in categories)
                {
                    dt.Rows.Add(c.Id, c.Name);
                }
                return dt;
            }
        }

        // Получить таблицу заказов
        public DataTable GetOrdersTable()
        {
            using (var uow = new UnitOfWork())
            {
                var dt = new DataTable();
                dt.Columns.Add("Id", typeof(int));
                dt.Columns.Add("OrderDate", typeof(DateTime));
                dt.Columns.Add("TotalAmount", typeof(decimal));

                var orders = uow.Orders.Query().ToList();
                foreach (var o in orders)
                {
                    dt.Rows.Add(o.Id, o.OrderDate, o.TotalAmount);
                }
                return dt;
            }
        }

BookStoreContext.cs
using System.Data.Entity;

namespace BookStore
{
    public class BookStoreContext : DbContext
    {
        public BookStoreContext() : base("name=BookStoreContext")
        {
            Database.SetInitializer(new CreateDatabaseIfNotExists<BookStoreContext>());
        }

        public DbSet<Book> Books { get; set; }
        public DbSet<Category> Categories { get; set; }
        public DbSet<User> Users { get; set; }
        public DbSet<Order> Orders { get; set; }
        public DbSet<OrderItem> OrderItems { get; set; }

        protected override void OnModelCreating(DbModelBuilder modelBuilder)
        {
            // Связь Книги и Категории
            modelBuilder.Entity<Book>()
                .HasOptional(b => b.CategoryEntity)
                .WithMany(c => c.Books)
                .HasForeignKey(b => b.CategoryId);

            // Связь Заказа и Пользователя
            modelBuilder.Entity<Order>()
                .HasOptional(o => o.User)
                .WithMany(u => u.Orders)
                .HasForeignKey(o => o.UserId);

            // Связь OrderItem с Order и Book
            modelBuilder.Entity<OrderItem>()
                .HasRequired(oi => oi.Order)
                .WithMany()
                .HasForeignKey(oi => oi.OrderId);

            modelBuilder.Entity<OrderItem>()
                .HasRequired(oi => oi.Book)
                .WithMany()
                .HasForeignKey(oi => oi.BookId);
        }
    }
}

RelayCommand.cs
using System;
using System.Windows.Input;

namespace BookStore
{
    public class RelayCommand : ICommand
    {
        private readonly Action<object> _execute;
        private readonly Func<object, bool> _canExecute;

        public RelayCommand(Action<object> execute, Func<object, bool> canExecute = null!)
        {
            _execute = execute;
            _canExecute = canExecute;
        }

        public event EventHandler? CanExecuteChanged
        {
            add { CommandManager.RequerySuggested += value; }
            remove { CommandManager.RequerySuggested -= value; }
        }

        public bool CanExecute(object parameter)
        {
            return _canExecute == null || _canExecute(parameter);
        }

        public void Execute(object parameter)
        {
            _execute(parameter);
        }
    }
}

CommandHistory.cs
using System.Collections.Generic;

namespace BookStore
{
    public interface IUndoRedoCommand
    {
        void Execute();
        void Undo();
    }

    public class AddBookCommand : IUndoRedoCommand
    {
        private readonly MainWindow _window;
        private readonly Book _book;

        public AddBookCommand(MainWindow window, Book book)
        {
            _window = window;
            _book = book;
        }

        public void Execute()
        {
            _window.AddBookDirect(_book);
        }

        public void Undo()
        {
            _window.RemoveBookDirect(_book.Id);
        }
    }

    public class RemoveBookCommand : IUndoRedoCommand
    {
        private readonly MainWindow _window;
        private readonly Book _book;

        public RemoveBookCommand(MainWindow window, Book book)
        {
            _window = window;
            _book = book;
        }

        public void Execute()
        {
            _window.RemoveBookDirect(_book.Id);
        }

        public void Undo()
        {
            _window.AddBookDirect(_book);
        }
    }

    public class EditBookCommand : IUndoRedoCommand
    {
        private readonly MainWindow _window;
        private readonly Book _oldBook;
        private readonly Book _newBook;

        public EditBookCommand(MainWindow window, Book oldBook, Book newBook)
        {
            _window = window;
            _oldBook = oldBook;
            _newBook = newBook;
        }

        public void Execute()
        {
            _window.EditBookDirect(_newBook);
        }

        public void Undo()
        {
            _window.EditBookDirect(_oldBook);
        }
    }

    public class CommandHistory
    {
        private Stack<IUndoRedoCommand> _undoStack = new Stack<IUndoRedoCommand>();
        private Stack<IUndoRedoCommand> _redoStack = new Stack<IUndoRedoCommand>();

        // Ссылка на главное окно для вызова обновления интерфейса
        private readonly MainWindow _window;

        // Конструктор теперь принимает MainWindow
        public CommandHistory(MainWindow window)
        {
            _window = window;
        }

        public void ExecuteCommand(IUndoRedoCommand command)
        {
            command.Execute();
            _undoStack.Push(command);
            _redoStack.Clear();

            // Обновляем фильтры после выполнения новой команды
            _window.FilterBooks();
        }

        public void Undo()
        {
            if (_undoStack.Count > 0)
            {
                var command = _undoStack.Pop();
                command.Undo();
                _redoStack.Push(command);
            }

            // Обновляем фильтры через ссылку на окно
            _window.FilterBooks();
        }

        public void Redo()
        {
            if (_redoStack.Count > 0)
            {
                var command = _redoStack.Pop();
                command.Execute();
                _undoStack.Push(command);
            }

            // Обновляем фильтры через ссылку на окно
            _window.FilterBooks();
        }

        public bool CanUndo => _undoStack.Count > 0;
        public bool CanRedo => _redoStack.Count > 0;
    }
}

using System;
using System.Globalization;
using System.IO;
using System.Windows.Data;
using System.Windows.Media.Imaging;

namespace BookStore
{
    public class BytesToImageConverter : IValueConverter
    {
        public object Convert(object value, Type targetType, object parameter, CultureInfo culture)
        {
            if (value is byte[] bytes && bytes.Length > 0)
            {
                try
                {
                    var image = new BitmapImage();
                    using (var mem = new MemoryStream(bytes))
                    {
                        mem.Position = 0;
                        image.BeginInit();
                        image.CreateOptions = BitmapCreateOptions.PreservePixelFormat;
                        image.CacheOption = BitmapCacheOption.OnLoad;
                        image.UriSource = null;
                        image.StreamSource = mem;
                        image.EndInit();
                    }
                    image.Freeze();
                    return image;
                }
                catch { return "Images/default-book.jpg"; }
            }

            return "/Images/default-book.jpg";
        }

        public object ConvertBack(object value, Type targetType, object parameter, CultureInfo culture)
            => throw new NotImplementedException();
    }
}

ImagePathConverter.cs
using System;
using System.Globalization;
using System.IO;
using System.Windows.Data;
using System.Windows.Media.Imaging;

namespace BookStore
{
    public class ImagePathConverter : IValueConverter
    {
        public object Convert(object value, Type targetType, object parameter, CultureInfo culture)
        {
            if (value is string imageName && !string.IsNullOrEmpty(imageName))
            {
                try
                {
                    // Вариант 1: Относительный путь к папке Images
                    string relativePath = $"Images/{imageName}";
                    if (File.Exists(relativePath))
                    {
                        return new BitmapImage(new Uri(relativePath, UriKind.Relative));
                    }

                    // Вариант 2: Абсолютный путь к папке Images в директории проекта
                    string absolutePath = System.IO.Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "Images", imageName);
                    if (File.Exists(absolutePath))
                    {
                        return new BitmapImage(new Uri(absolutePath));
                    }

                    // Вариант 3: Ищем в папке проекта
                    string projectPath = System.IO.Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "..", "..", "..", "Images", imageName);
                    if (File.Exists(projectPath))
                    {
                        return new BitmapImage(new Uri(projectPath));
                    }
                }
                catch (Exception ex)
                {
                    System.Diagnostics.Debug.WriteLine($"Ошибка загрузки изображения {imageName}: {ex.Message}");
                }
            }

            // Загружаем изображение по умолчанию
            try
            {
                string defaultImagePath = "Images/default-book.jpg";
                if (File.Exists(defaultImagePath))
                {
                    return new BitmapImage(new Uri(defaultImagePath, UriKind.Relative));
                }

                string absoluteDefaultPath = System.IO.Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "Images", "default-book.jpg");
                if (File.Exists(absoluteDefaultPath))
                {
                    return new BitmapImage(new Uri(absoluteDefaultPath));
                }
            }
            catch (Exception ex)
            {
                System.Diagnostics.Debug.WriteLine($"Ошибка загрузки изображения по умолчанию: {ex.Message}");
            }

            return null!;
        }

        public object ConvertBack(object value, Type targetType, object parameter, CultureInfo culture)
        {
            throw new NotImplementedException();
        }
    }
}
